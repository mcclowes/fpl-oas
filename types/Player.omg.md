# Player (Element)

A player in the Fantasy Premier League, also referred to as an "element" in the API.

```omg.type
type Player = {
  id: integer @description("Player ID used by FPL endpoints"),
  code: integer @description("Player ID in the Premier League's data service"),
  first_name: string @description("Player's first name"),
  second_name: string @description("Player's surname"),
  web_name: string @description("Short display name used by FPL"),
  team: integer @description("Current Premier League team ID"),
  team_code: integer @description("Current team's external data-service code"),
  element_type: integer @description("Position: 1=GKP, 2=DEF, 3=MID, 4=FWD"),
  status: "a" | "d" | "i" | "n" | "s" | "u" @description("a=available, d=doubtful, i=injured, n=not available, s=suspended, u=unavailable"),
  now_cost: integer @description("Current price in 0.1m units (e.g., 100 = 10.0m)"),
  cost_change_start: integer @description("Price change since the start of the season, in 0.1m units"),
  cost_change_event: integer @description("Price change during the current gameweek, in 0.1m units"),
  cost_change_start_fall: integer @description("Season price fall expressed as a positive number in 0.1m units"),
  cost_change_event_fall: integer @description("Current-gameweek price fall expressed as a positive number in 0.1m units"),
  photo: string @description("Player photo filename used by FPL's image service"),
  news: string @description("Latest availability note; empty when there is no news"),
  news_added: string? @format("date-time") @description("Time the availability note was last updated, or null when there is no news"),
  chance_of_playing_next_round: integer? @min(0) @max(100) @description("Estimated chance of playing next gameweek as a percentage, or null when no estimate is needed"),
  chance_of_playing_this_round: integer? @min(0) @max(100) @description("Estimated chance of playing this gameweek as a percentage, or null when no estimate is needed"),
  selected_by_percent: string @description("Percentage of squads that own the player, encoded as a decimal string"),
  form: string @description("Recent points-weighted form, encoded as a decimal string"),
  points_per_game: string @description("Average FPL points per appearance, encoded as a decimal string"),
  ep_this: string? @description("Expected points this gameweek"),
  ep_next: string? @description("Expected points next gameweek"),
  value_form: string @description("Form divided by price, encoded as a decimal string"),
  value_season: string @description("Season points divided by price, encoded as a decimal string"),
  total_points: integer @description("Total FPL points this season"),
  event_points: integer @description("FPL points in the current gameweek"),
  transfers_in: integer @description("Total transfers in this season"),
  transfers_out: integer @description("Total transfers out this season"),
  transfers_in_event: integer @description("Transfers in during the current gameweek"),
  transfers_out_event: integer @description("Transfers out during the current gameweek"),
  in_dreamteam: boolean @description("Whether the player is in the season dream team"),
  dreamteam_count: integer @description("Number of gameweek dream teams the player has appeared in"),
  special: boolean @description("Whether the player is a special element governed by separate rules"),
  removed: boolean @description("Whether the player has been removed from the game"),
  can_transact: boolean @description("Whether managers can currently buy or sell the player"),
  can_select: boolean @description("Whether managers can select the player in a squad"),
  squad_number: integer? @description("Premier League squad number, or null when unassigned"),
  region: integer? @description("Player nationality region ID, or null when unavailable"),
  team_join_date: string? @format("date") @description("Date the player joined the current club, or null when unavailable"),
  birth_date: string? @format("date") @description("Player's date of birth, or null when unavailable"),
  has_temporary_code: boolean @description("Whether the external player code is temporary"),
  opta_code: string? @description("Opta player identifier, or null when unavailable"),
  minutes: integer @description("Minutes played this season"),
  goals_scored: integer @description("Premier League goals scored this season"),
  assists: integer @description("FPL assists awarded this season"),
  clean_sheets: integer @description("FPL clean sheets awarded this season"),
  goals_conceded: integer @description("Goals conceded while the player was on the pitch"),
  own_goals: integer @description("Own goals scored this season"),
  penalties_saved: integer @description("Penalties saved this season"),
  penalties_missed: integer @description("Penalties missed this season"),
  yellow_cards: integer @description("Yellow cards received this season"),
  red_cards: integer @description("Red cards received this season"),
  saves: integer @description("Goalkeeper saves this season"),
  bonus: integer @description("Bonus points awarded this season"),
  bps: integer @description("Bonus Points System score"),
  influence: string @description("ICT influence score, encoded as a decimal string"),
  creativity: string @description("ICT creativity score, encoded as a decimal string"),
  threat: string @description("ICT threat score, encoded as a decimal string"),
  ict_index: string @description("ICT Index combining influence, creativity, and threat"),
  starts: integer @description("Premier League starts this season"),
  expected_goals: string @description("Expected goals this season, encoded as a decimal string"),
  expected_assists: string @description("Expected assists this season, encoded as a decimal string"),
  expected_goal_involvements: string @description("Expected goals plus expected assists, encoded as a decimal string"),
  expected_goals_conceded: string @description("Team expected goals conceded while the player was on the pitch, encoded as a decimal string"),
  influence_rank: integer @description("Overall rank by influence score"),
  influence_rank_type: integer @description("Rank by influence score among players in the same position"),
  creativity_rank: integer @description("Overall rank by creativity score"),
  creativity_rank_type: integer @description("Rank by creativity score among players in the same position"),
  threat_rank: integer @description("Overall rank by threat score"),
  threat_rank_type: integer @description("Rank by threat score among players in the same position"),
  ict_index_rank: integer @description("Overall rank by ICT Index"),
  ict_index_rank_type: integer @description("Rank by ICT Index among players in the same position"),
  corners_and_indirect_freekicks_order: integer? @description("Team priority for corners and indirect free kicks, or null when not listed"),
  corners_and_indirect_freekicks_text: string @description("Human-readable corner and indirect-free-kick role; usually empty"),
  direct_freekicks_order: integer? @description("Team priority for direct free kicks, or null when not listed"),
  direct_freekicks_text: string @description("Human-readable direct-free-kick role; usually empty"),
  penalties_order: integer? @description("Team penalty-taking priority, or null when not listed"),
  penalties_text: string @description("Human-readable penalty-taking role; usually empty"),
  expected_goals_per_90: number @description("Expected goals per 90 minutes"),
  saves_per_90: number @description("Goalkeeper saves per 90 minutes"),
  expected_assists_per_90: number @description("Expected assists per 90 minutes"),
  expected_goal_involvements_per_90: number @description("Expected goals plus expected assists per 90 minutes"),
  expected_goals_conceded_per_90: number @description("Team expected goals conceded per 90 minutes while the player was on the pitch"),
  goals_conceded_per_90: number @description("Goals conceded per 90 minutes while the player was on the pitch"),
  now_cost_rank: integer @description("Overall rank from most to least expensive"),
  now_cost_rank_type: integer @description("Price rank among players in the same position"),
  form_rank: integer @description("Overall rank by recent form"),
  form_rank_type: integer @description("Recent-form rank among players in the same position"),
  points_per_game_rank: integer @description("Overall rank by points per game"),
  points_per_game_rank_type: integer @description("Points-per-game rank among players in the same position"),
  selected_rank: integer @description("Overall rank by squad ownership"),
  selected_rank_type: integer @description("Ownership rank among players in the same position"),
  starts_per_90: number @description("Starts per 90 minutes played"),
  clean_sheets_per_90: number @description("Clean sheets per 90 minutes played")
}
```
